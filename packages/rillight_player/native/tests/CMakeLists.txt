cmake_minimum_required(VERSION 3.21)
project(rillight_surface_test LANGUAGES CXX)
include(../dependencies.cmake)
if(NOT FLUTTER_ENGINE)
  message(FATAL_ERROR "Set FLUTTER_ENGINE to flutter/bin/cache/artifacts/engine/windows-x64")
endif()
add_executable(surface_test surface_test.cpp ../../windows/video_surface.cpp)
add_library(plugin_compile OBJECT ../../windows/rillight_player_plugin.cpp)
target_compile_features(plugin_compile PRIVATE cxx_std_17)
target_compile_definitions(plugin_compile PRIVATE FLUTTER_PLUGIN_IMPL NOMINMAX _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR)
target_include_directories(plugin_compile PRIVATE "${FLUTTER_ENGINE}" "${FLUTTER_ENGINE}/cpp_client_wrapper/include" "${RILLIGHT_MPV}/include/mpv" "${RILLIGHT_MPV}/include" "${RILLIGHT_ANGLE}/include")
target_compile_features(surface_test PRIVATE cxx_std_17)
target_compile_definitions(surface_test PRIVATE NOMINMAX _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR)
target_include_directories(surface_test PRIVATE "${FLUTTER_ENGINE}" "${FLUTTER_ENGINE}/cpp_client_wrapper/include" "${RILLIGHT_MPV}/include/mpv" "${RILLIGHT_MPV}/include" "${RILLIGHT_ANGLE}/include")
target_link_libraries(surface_test PRIVATE d3d11 dxgi "${RILLIGHT_MPV}/libmpv.dll.a" "${RILLIGHT_ANGLE}/lib/libEGL.dll.lib" "${RILLIGHT_ANGLE}/lib/libGLESv2.dll.lib")
file(GLOB RUNTIME "${RILLIGHT_ANGLE}/*.dll" "${RILLIGHT_MPV}/*.dll")
foreach(DLL ${RUNTIME})
  add_custom_command(TARGET surface_test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${DLL}" "$<TARGET_FILE_DIR:surface_test>")
endforeach()
